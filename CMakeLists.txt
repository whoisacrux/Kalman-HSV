cmake_minimum_required(VERSION 3.20)
project(HSVKalmanLab VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BUILD_LAB "Build the OpenCV desktop application" ON)
include(CTest)
if(BUILD_LAB)
  find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc highgui videoio)
  add_executable(hsv_kalman_lab src/main.cpp)
  target_include_directories(hsv_kalman_lab PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(hsv_kalman_lab PRIVATE ${OpenCV_LIBS})
  if(MSVC)
    target_compile_options(hsv_kalman_lab PRIVATE /W4 /utf-8)
  else()
    target_compile_options(hsv_kalman_lab PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()
if(BUILD_TESTING)
  add_executable(kalman_tests tests/kalman_tests.cpp)
  target_include_directories(kalman_tests PRIVATE src)
  add_test(NAME scalar_kalman COMMAND kalman_tests)
  add_executable(motion_tests tests/motion_tests.cpp)
  target_include_directories(motion_tests PRIVATE src)
  add_test(NAME synthetic_motion COMMAND motion_tests)
endif()
